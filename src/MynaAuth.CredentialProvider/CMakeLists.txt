cmake_minimum_required(VERSION 3.24)
project(MynaAuthCredentialProvider LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(MynaAuthCredentialProvider SHARED Provider.cpp Provider.def)
target_compile_definitions(MynaAuthCredentialProvider PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MynaAuthCredentialProvider PRIVATE /W4 /WX /utf-8 /guard:cf /sdl)
target_link_options(MynaAuthCredentialProvider PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(MynaAuthCredentialProvider PRIVATE credui secur32 advapi32 ole32 uuid gdi32 shlwapi)
add_executable(ProviderChecks ProviderChecks.cpp)
target_compile_definitions(ProviderChecks PRIVATE UNICODE _UNICODE)
target_compile_options(ProviderChecks PRIVATE /utf-8 /W4 /WX)
target_link_libraries(ProviderChecks PRIVATE credui secur32 advapi32 ole32 uuid gdi32 shlwapi)
